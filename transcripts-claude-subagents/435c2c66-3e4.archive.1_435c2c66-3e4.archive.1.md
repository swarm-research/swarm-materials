import sys, json, math, collections
from collections import defaultdict, Counter

# Load clean citations
cits = []
real_agents = set(f'agent-{i:03d}' for i in range(1, 37))
with open('citations.jsonl') as f:
    for line in f:
        r = json.loads(line.strip())
        citer = r.get('citer','')
        cited = r.get('cited','')
        file_field = str(r.get('file',''))
        if citer in real_agents and cited in real_agents and 'xxxx' not in file_field and not file_field.startswith('{'):
            cits.append(r)

print(f'Clean citations: {len(cits)}', flush=True)

# Build adjacency matrix
adj = defaultdict(lambda: Counter())
for c in cits:
    adj[c['citer']][c['cited']] += 1

# 1. Degree distribution
in_deg = Counter()
out_deg = Counter()
for c in cits:
    in_deg[c['cited']] += 1
    out_deg[c['citer']] += 1

unique_in = defaultdict(set)
unique_out = defaultdict(set)
for c in cits:
    unique_in[c['cited']].add(c['citer'])
    unique_out[c['citer']].add(c['cited'])

print(f'Agents with in-deg > 0: {len(unique_in)}')
print(f'Agents with out-deg > 0: {len(unique_out)}')
print(f'Agents in network: {len(set(list(unique_in.keys()) + list(unique_out.keys())))}')

# Gini coefficient of in-degree (unique)
in_vals = sorted([len(v) for v in unique_in.values()], reverse=True)
n = len(in_vals)
gini = 0
for i, val in enumerate(in_vals):
    gini += (2 * (i+1) - n - 1) * val
gini = gini / (n * sum(in_vals)) if sum(in_vals) > 0 else 0
print(f'Gini coefficient (unique citers): {gini:.3f}')

# Top 10 share
top10_share = sum(in_vals[:10]) / sum(in_vals) * 100 if sum(in_vals) > 0 else 0
print(f'Top 10 share of all citations: {top10_share:.1f}%')

# 2. Reciprocity
mutual_pairs = 0
total_pairs = 0
agents = set(unique_in.keys()) | set(unique_out.keys())
for a in agents:
    for b in agents:
        if a < b:  # count each pair once
            a_to_b = b in unique_out.get(a, set())
            b_to_a = a in unique_out.get(b, set())
            if a_to_b or b_to_a:
                total_pairs += 1
                if a_to_b and b_to_a:
                    mutual_pairs += 1

reciprocity = mutual_pairs / max(total_pairs, 1) * 100
print(f'Reciprocity: {mutual_pairs}/{total_pairs} = {reciprocity:.1f}%')

# 3. Self-citation rate
self_cites = sum(1 for c in cits if c['citer'] == c['cited'])
print(f'Self-citations: {self_cites} ({self_cites/len(cits)*100:.1f}%)')

# 4. PageRank
damping = 0.85
pr = {a: 1.0/len(agents) for a in agents}
for _ in range(100):
    new_pr = {}
    for a in agents:
        rank = (1 - damping) / len(agents)
        for b in agents:
            out_links = len(unique_out.get(b, set()))
            if out_links > 0 and a in unique_out.get(b, set()):
                rank += damping * pr[b] / out_links
        new_pr[a] = rank
    pr = new_pr

pr_ranked = sorted(pr.items(), key=lambda x: -x[1])
print()
print('PageRank Top 15:')
for i, (a, s) in enumerate(pr_ranked[:15], 1):
    print(f'  {i:2d}. {a}: {s*100:.2f}% (in: {len(unique_in.get(a,set()))}, out: {len(unique_out.get(a,set()))})')

# 5. Compare PageRank vs raw in-degree
in_ranked = sorted(unique_in.items(), key=lambda x: -len(x[1]))
pr_dict = {a: i for i, (a, _) in enumerate(pr_ranked)}
in_dict = {a: i for i, (a, _) in enumerate(in_ranked)}

# Compute rank correlation (Spearman's rho approximation)
common = set(pr_dict.keys()) & set(in_dict.keys())
n = len(common)
diffs = sum((pr_dict[a] - in_dict[a])**2 for a in common)
rho = 1 - (6 * diffs) / (n * (n*n - 1)) if n > 2 else 0
print(f'Spearman rho (PageRank vs raw in-degree): {rho:.3f}')

# 6. Clustering coefficient (average)
# For each node, what fraction of its neighbors are connected to each other?
def clustering_coeff(node, adj_out, adj_in):
    neighbors = set(adj_out.get(node, set())) | set(adj_in.get(node, set()))
    k = len(neighbors)
    if k < 2:
        return 0
    links = 0
    n_list = list(neighbors)
    for i in range(len(n_list)):
        for j in range(i+1, len(n_list)):
            a, b = n_list[i], n_list[j]
            if b in adj_out.get(a, set()) or a in adj_out.get(b, set()):
                links += 1
    return (2 * links) / (k * (k - 1))

adj_out = unique_out
adj_in = unique_in

clust_vals = []
for a in agents:
    cc = clustering_coeff(a, adj_out, adj_in)
    clust_vals.append(cc)

avg_clust = sum(clust_vals) / len(clust_vals) if clust_vals else 0
print(f'Average clustering coefficient: {avg_clust:.3f}')

# 7. Network density
possible = len(agents) * (len(agents) - 1)
actual = sum(len(v) for v in unique_out.values())
density = actual / possible * 100
print(f'Network density: {actual}/{possible} = {density:.2f}%')
